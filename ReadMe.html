    // 搜尋輸入監聽
    function handleSearchInput() {

    function triggerDeepProbe() {
      const kw = document.getElementById('search-input').value.trim().toLowerCase();
      if (!kw) {
        alert('請先在搜尋框輸入要探測的單字名稱（例如 volcano 或 hesitate）');
        return;
      }
      probeWordAcrossDirectories(kw);
    }

    // 核心探針：同時探測子目錄 (A1/, A2/, B1/...) 與根目錄
    async function probeWordAcrossDirectories(word) {
      const candidatePaths = [
        `A1/${word}.html`,
        `A2/${word}.html`,
        `B1/${word}.html`,
        `B2/${word}.html`,
        `C1/${word}.html`,
        `C2/${word}.html`,
        `${word}.html`
      ];

      for (let path of candidatePaths) {
        const found = await testPathExistence(path);
        if (found) {
          showDiscovery(word, path, found.cefr, found.defZh, found.pos, found.curriculum);
          return;
        }
      }
    }

    // 透過隱藏 iframe 探測路徑與解析 Meta 中繼標籤
    function testPathExistence(relPath) {
      return new Promise((resolve) => {
        const frame = document.getElementById('probe-frame');
        let resolved = false;

        const cleanup = () => {
          frame.onload = null;
          frame.onerror = null;
        };

        frame.onload = () => {
          if (resolved) return;
          resolved = true;
          try {
            const doc = frame.contentDocument || frame.contentWindow.document;
            // 嘗試讀取 eng-vocab 規範的中繼標籤
            const metaCefr = doc.querySelector('meta[name="vocab-cefr"]')?.content;
            const metaDef = doc.querySelector('meta[name="vocab-def"]')?.content;
            const metaPos = doc.querySelector('meta[name="vocab-pos"]')?.content;
            const metaCurr = doc.querySelector('meta[name="vocab-curriculum"]')?.content;

            // 若無 meta，嘗試抓取 title 作為備用語意
            const title = doc.title || '';
            const inferredCefr = metaCefr || (relPath.includes('/') ? relPath.split('/')[0] : 'B1');
            const inferredDef = metaDef || '（已偵測到檔案，點擊加入以同步）';

            cleanup();
            resolve({
              cefr: inferredCefr,
              defZh: inferredDef,
              pos: metaPos || '',
              curriculum: metaCurr || ''
            });
          } catch (e) {
            // 本機受限時，若成功載入仍視為檔案存在
            const inferredCefr = relPath.includes('/') ? relPath.split('/')[0] : 'B1';
            cleanup();
            resolve({ cefr: inferredCefr, defZh: '本地檔案已就緒', pos: '', curriculum: '' });
          }
        };

        frame.onerror = () => {
          if (resolved) return;
          resolved = true;
          cleanup();
          resolve(null);
        };

        // 觸發載入
        frame.src = './' + relPath + '?t=' + Date.now();

        // 700ms 超時保護
        setTimeout(() => {
          if (!resolved) {
            resolved = true;
            cleanup();
            resolve(null);
          }
        }, 700);
      });
    }

    // 加入索引庫並觸發雙軌儲存
    async function confirmAddToDatabase() {

      // 避免重複

    // 寫入檔案 (方案2 核心)
    async function writeDatabaseWithFileSystem(handle) {
      try {
        const fileHandle = await handle.getFileHandle('vocab-db.js', { create: true });
        const writable = await fileHandle.createWritable();
        const content = generateDbFileContent();
        await writable.write(content);
        await writable.close();
      } catch (err) {
        console.error('寫入失敗，改用下載提示:', err);
      }
    }

    // 方案 1 產生 JS 內容並下載
    function exportDatabaseFile() {
      const content = generateDbFileContent();
      const blob = new Blob([content], { type: 'application/javascript;charset=utf-8' });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = 'vocab-db.js';
      document.body.appendChild(a);
      a.click();
      document.body.removeChild(a);
      URL.revokeObjectURL(url);
    }

    function generateDbFileContent() {

📁 單字庫總目錄/
├── 📄 index.html        <--- 貼入上方步驟二的程式碼
├── 📄 vocab-db.js       <--- 貼入上方步驟一的程式碼

  1. 打開即用：
  • 雙擊開啟 index.html，所有單字卡與 CEFR 級別篩選立即流暢呈現。
  1. 體驗「自動探針與新單字提示」：

  • 當您使用 @eng-vocab 產出一個新單字（如 approach.html）並存入 B1/ 子目錄後：

  • 回到 index.html 的搜尋框輸入 approach，系統探針會在幾百毫秒內自動掃描到 B1/approach.html，並在頂部彈出醒目的橘黃色提示橫幅：

    💡 在目錄中偵測到未收錄單字：approach 路徑：B1/approach.html ｜ 級別：CEFR B1 [➕ 立即加入索引資料庫]
